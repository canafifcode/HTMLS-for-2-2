STATE (with a temporary-override twist)

A1 — Brain-Support Subscription Simulator

Three tiers (Common/Plus/Lux) with different coverage radius, promote()/demote() transitions, a temporary activateLux(hours) that auto-reverts, and setMood() gated on Lux being active.
Why State Every operation's behavior — the safe travel radius, whether mood control is available — depends entirely on which tier is currently active, and that tier changes over the object's lifetime (promote/demote/temporary-Lux). That's the textbook State signature: instead of an if(tier==COMMON) ... else if(tier==PLUS) ... ladder inside every method, each tier becomes its own class implementing the shared behavior. The twist — activateLux(hours) temporarily overriding the state and then restoring the previous one — is a common real-world extension of State: the context simply remembers which state to snap back to, the same idea as a "return to previous screen" stack. 
// ---------- TierState.java ----------
public interface TierState {
    String getName();
    int getRangeKm();
    void promote(PatientMonitor ctx);
    void demote(PatientMonitor ctx);
}

// ---------- CommonState.java ----------
public class CommonState implements TierState {
    public String getName() { return "Common"; }
    public int getRangeKm() { return 10; }
    public void promote(PatientMonitor ctx) {
        ctx.setTier(ctx.getPlusState());
        System.out.println("[promote] Common -> Plus");
    }
    public void demote(PatientMonitor ctx) {
        System.out.println("[demote] Already at Common. No change.");
    }
}

// ---------- PlusState.java ----------
public class PlusState implements TierState {
    public String getName() { return "Plus"; }
    public int getRangeKm() { return 50; }
    public void promote(PatientMonitor ctx) {
        ctx.setTier(ctx.getLuxState());
        System.out.println("[promote] Plus -> Lux");
    }
    public void demote(PatientMonitor ctx) {
        ctx.setTier(ctx.getCommonState());
        System.out.println("[demote] Plus -> Common");
    }
}

// ---------- LuxState.java ----------
public class LuxState implements TierState {
    public String getName() { return "Lux"; }
    public int getRangeKm() { return 50; }
    public void promote(PatientMonitor ctx) {
        System.out.println("[promote] Already at Lux. No change.");
    }
    public void demote(PatientMonitor ctx) {
        ctx.setTier(ctx.getPlusState());
        System.out.println("[demote] Lux -> Plus");
    }
}

// ---------- PatientMonitor.java (Context) ----------
public class PatientMonitor {
    private final TierState common = new CommonState();
    private final TierState plus   = new PlusState();
    private final TierState lux    = new LuxState();
    private TierState currentTier = common;      // starts at Common

    TierState getCommonState() { return common; }
    TierState getPlusState()   { return plus; }
    TierState getLuxState()    { return lux; }
    void setTier(TierState t) { currentTier = t; }

    // ---- 1. travelCheck(km) ----
    public void travelCheck(int km) {
        if (km <= currentTier.getRangeKm()) {
            System.out.println("[travelCheck(" + km + "km)] " + currentTier.getName() + " tier -> STABLE");
        } else {
            System.out.println("[travelCheck(" + km + "km)] " + currentTier.getName()
                + " tier -> UNSTABLE! ALERT: patient has blacked out -- bring the patient back into coverage now.");
            System.out.println("   -> Bringing patient back into range...");
            travelCheck(0);                              // simulation continues with travelCheck(0)
            System.out.println("   -> Patient regains consciousness.");
        }
    }

    // ---- 2. activateLux(hours) — temporary override with auto-revert ----
    private TierState preLuxTier = null;
    public void activateLux(int hours) {
        if (currentTier == lux) {
            System.out.println("[activateLux] Already on Lux. Nothing to override.");
            return;
        }
        preLuxTier = currentTier;
        currentTier = lux;
        System.out.println("[activateLux] Temporarily switching to Lux for " + hours + " hour(s)...");
        final TierState toRestore = preLuxTier;
        // background timer thread -- keeps the demo non-blocking so we can
        // call setMood() WHILE Lux is still active, then watch it auto-revert
        new Thread(() -> {
            try { Thread.sleep(hours * 300L); } catch (InterruptedException ignored) {}
            currentTier = toRestore;
            preLuxTier = null;
            System.out.println("[activateLux] Duration elapsed. Reverted to " + toRestore.getName() + " tier.");
        }).start();
    }

    // ---- 3. setMood(...) — only while Lux is active ----
    public void setMood(String mood) {
        if (currentTier == lux) System.out.println("[setMood] Mood set to: " + mood);
        else System.out.println("[setMood] Mood control unavailable.");
    }

    // ---- 4 & 5. promote() / demote() — delegate to current state ----
    public void promote() { currentTier.promote(this); }
    public void demote() { currentTier.demote(this); }
}

// ---------- Demo.java ----------
public class Demo {
    public static void main(String[] args) throws InterruptedException {
        PatientMonitor monitor = new PatientMonitor();

        monitor.travelCheck(5);        // Common, in range
        monitor.travelCheck(15);       // Common, OUT of range -> alert + auto-recover
        monitor.promote();             // Common -> Plus
        monitor.travelCheck(30);       // Plus, in range
        monitor.setMood("calm");     // not Lux -> unavailable

        monitor.activateLux(2);        // temporary Lux for ~2 units
        monitor.setMood("happy");    // Lux active -> succeeds
        Thread.sleep(900);             // let the background revert finish
        monitor.setMood("exhausted"); // reverted to Plus -> unavailable again

        monitor.demote();              // Plus -> Common
        monitor.demote();              // Common -> Common (no change)
    }
}

ALERT: patient has blacked out -- bring the patient back into coverage now. -> Bringing patient back into range... [travelCheck(0km)] Common tier -> STABLE -> Patient regains consciousness. [promote] Common -> Plus [travelCheck(30km)] Plus tier -> STABLE [setMood] Mood control unavailable. [activateLux] Temporarily switching to Lux for 2 hour(s)... OBSERVER

A2 — RavenBoard

Multiple groups (Commander, Scouts, Supply Team) receive new scroll messages and react; groups can subscribe/unsubscribe at runtime.
Why Observer This is the magazine-subscription story from lecture, reskinned as ravens: one publisher (the board), many subscribers who each react in their own way, and a runtime subscribe/unsubscribe list. The board never needs to know what Scouts or the Supply Team do with a message — it just calls receive() on whoever is currently registered. That decoupling plus the explicit "subscribe/unsubscribe at runtime" phrase are the two strongest Observer tells. 
// ---------- Subscriber.java ----------
public interface Subscriber {
    String getName();
    void receive(String message);
}

// ---------- Commander.java / Scouts.java / SupplyTeam.java ----------
public class Commander implements Subscriber {
    public String getName() { return "Commander"; }
    public void receive(String message) {
        System.out.println("[Commander] Noted: \"" + message + "\"");
    }
}
public class Scouts implements Subscriber {
    public String getName() { return "Scouts"; }
    public void receive(String message) {
        System.out.println("[Scouts] " + message +  -> Dispatch riders!");
    }
}
public class SupplyTeam implements Subscriber {
    public String getName() { return "Supply Team"; }
    public void receive(String message) {
        System.out.println("[Supply Team] " + message +  -> Update inventory!");
    }
}

// ---------- RavenBoard.java (Subject) ----------
import java.util.*;
public class RavenBoard {
    private List<Subscriber> subscribers = new ArrayList<>();

    public void subscribe(Subscriber s) {
        subscribers.add(s);
        System.out.println("[RavenBoard] " + s.getName() + " has joined the board room.");
    }
    public void unsubscribe(Subscriber s) {
        subscribers.remove(s);
        System.out.println("[RavenBoard] " + s.getName() + " has left the board room.");
    }
    public void postMessage(String message) {
        System.out.println("\n[RavenBoard] New scroll: \"" + message + "\"");
        for (Subscriber s : subscribers) s.receive(message);
    }
}

// ---------- Demo.java ----------
public class Demo {
    public static void main(String[] args) {
        RavenBoard board = new RavenBoard();
        Subscriber commander = new Commander();
        Subscriber scouts    = new Scouts();
        Subscriber supply    = new SupplyTeam();

        board.subscribe(commander);
        board.subscribe(scouts);
        board.subscribe(supply);

        board.postMessage("Enemy spotted near the river");

        board.unsubscribe(scouts);                       // Scouts leave the board room
        board.postMessage("Winter supplies running low");

        board.subscribe(scouts);                          // Scouts rejoin at runtime
        board.postMessage("Ships seen in the east");
    }
}

MEDIATOR

B2 — Smart Home Automation Hub

Light Sensor, Blinds and Air Conditioner must not talk to each other directly; a Central Hub coordinates the chain reaction between them.
Why Mediator The requirement states this outright: "the devices should not talk to each other directly... they all report to the Central Hub." That's the air-traffic-control structure exactly — each device knows only the Hub, and the Hub decides what happens next, including chaining a second reaction (Blinds close → Hub tells AC to turn on). Notice this is Mediator and not Observer: the Hub doesn't just broadcast a plain notification to a list of subscribers, it actively interprets each event and issues specific commands to specific other devices — coordination logic lives centrally, which is the Mediator's defining trait. 
// ---------- Mediator.java ----------
public interface Mediator {
    void notify(Object sender, String event);
}

// ---------- LightSensor.java ----------
public class LightSensor {
    private Mediator hub;
    public LightSensor(Mediator hub) { this.hub = hub; }
    public void detectHighBrightness() {
        System.out.println("[LightSensor] High brightness detected.");
        hub.notify(this, "HIGH_BRIGHTNESS");      // no direct call to Blinds
    }
}

// ---------- Blinds.java ----------
public class Blinds {
    private Mediator hub;
    public Blinds(Mediator hub) { this.hub = hub; }
    public void close() {
        System.out.println("[Blinds] Closing.");
        hub.notify(this, "BLINDS_CLOSED");          // no direct call to AC
    }
}

// ---------- AirConditioner.java ----------
public class AirConditioner {
    public void turnOn() {
        System.out.println("[AirConditioner] Turning ON -- cooling the room.");
    }
}

// ---------- SmartHomeHub.java (concrete Mediator) ----------
public class SmartHomeHub implements Mediator {
    private Blinds blinds;
    private AirConditioner ac;

    public void register(Blinds b, AirConditioner a) { this.blinds = b; this.ac = a; }

    public void notify(Object sender, String event) {
        if (sender instanceof LightSensor && event.equals("HIGH_BRIGHTNESS")) {
            System.out.println("[Hub] High brightness -> closing blinds.");
            blinds.close();
        } else if (sender instanceof Blinds && event.equals("BLINDS_CLOSED")) {
            System.out.println("[Hub] Blinds closed -> turning on AC.");
            ac.turnOn();
        }
    }
}

// ---------- Demo.java ----------
public class Demo {
    public static void main(String[] args) {
        SmartHomeHub hub = new SmartHomeHub();
        LightSensor sensor = new LightSensor(hub);
        Blinds blinds = new Blinds(hub);
        AirConditioner ac = new AirConditioner();
        hub.register(blinds, ac);

        sensor.detectHighBrightness();   // triggers the whole chain through the hub
    }
}
Sample output[LightSensor] High brightness detected. [Hub] High brightness -> closing blinds. [Blinds] Closing. [Hub] Blinds closed -> turning on AC. [AirConditioner] Turning ON -- cooling the room.
TEMPLATE METHOD

C1 — Hospital Visit Simulator

Check-In → Vitals → Assessment → Treatment → Discharge is the same 5-step flow for every department, but Assessment and Treatment differ per department.
Why Template Method This is your Coffee/Tea slide with a stethoscope. The order and count of steps never changes — that invariant skeleton belongs in an abstract superclass as a final method. Check-In, Record Vitals and Discharge Summary are identical everywhere, so they stay concrete in the base class; Assessment and Treatment are declared abstract and supplied per department. If instead each department picked a whole different algorithm with no shared skeleton, you'd reach for Strategy — but here the structure is fixed and only two steps vary, which is exactly Template Method's territory. 
// ---------- Department.java (abstract class with the template method) ----------
public abstract class Department {

    public final void visit(String patientName, int visitId) {   // the fixed skeleton
        checkIn(patientName, visitId);
        recordVitals();
        assessment();          // abstract step
        treatment();           // abstract step
        dischargeSummary();
        System.out.println();
    }

    private void checkIn(String name, int id) {
        System.out.println("[Check-In] " + name + " registered, Visit ID #" + id);
    }
    private void recordVitals() {
        System.out.println("[Vitals] Temperature: 98.6F, Blood Pressure: 120/80");
    }
    protected abstract void assessment();
    protected abstract void treatment();
    private void dischargeSummary() {
        System.out.println("[Discharge] Patient discharged. Notes filed in patient record.");
    }
}

// ---------- GeneralDepartment.java ----------
public class GeneralDepartment extends Department {
    protected void assessment() { System.out.println("[Assessment] Doctor performs normal diagnosis"); }
    protected void treatment()  { System.out.println("[Treatment] Prescribe standard medicine"); }
}

// ---------- PediatricsDepartment.java ----------
public class PediatricsDepartment extends Department {
    protected void assessment() { System.out.println("[Assessment] Doctor checks symptoms while ensuring child comfort level"); }
    protected void treatment()  { System.out.println("[Treatment] Give child-safe medicine, friendly reassurance message"); }
}

// ---------- EmergencyDepartment.java ----------
public class EmergencyDepartment extends Department {
    protected void assessment() { System.out.println("[Assessment] Quick triage check (urgent/non-urgent)"); }
    protected void treatment()  { System.out.println("[Treatment] Immediate emergency procedure"); }
}

// ---------- Demo.java ----------
public class Demo {
    public static void main(String[] args) {
        Department general    = new GeneralDepartment();
        Department pediatrics = new PediatricsDepartment();
        Department emergency  = new EmergencyDepartment();

        System.out.println("=== General Department ===");
        general.visit("Rahim", 101);
        System.out.println("=== Pediatrics Department ===");
        pediatrics.visit("Tumpa", 102);
        System.out.println("=== Emergency Department ===");
        emergency.visit("Karim", 103);
    }
}

OBSERVER

C2 — Financial Dashboard (Stock Feed)

StockData feed must notify Ticker Tape, Graph and Buy/Sell Bot automatically on price change, without knowing their specifics, and support adding/removing widgets.
Why Observer Textbook Weather-o-Rama, just with candlesticks instead of clouds. The phrase to underline in the question is "StockData feed doesn't know the specifics of these widgets but can notify all of them automatically" — that's the Subject/Observer decoupling by definition, and "provision of adding or removing widgets" is literally registerObserver/removeObserver. Don't be tempted by Mediator here: the widgets never coordinate with each other or need central arbitration — they all just react independently to the one subject's state change, which is precisely Observer's one-way, one-to-many shape. 
// ---------- StockObserver.java ----------
public interface StockObserver {
    void onPriceUpdate(String symbol, double price);
}

// ---------- TickerTape.java ----------
public class TickerTape implements StockObserver {
    public void onPriceUpdate(String symbol, double price) {
        System.out.println("[TickerTape] scrolling " + symbol + " @ $" + price);
    }
}

// ---------- GraphWidget.java ----------
public class GraphWidget implements StockObserver {
    public void onPriceUpdate(String symbol, double price) {
        System.out.println("[Graph] plotting new point for " + symbol + ": " + price);
    }
}

// ---------- TradeBot.java ----------
public class TradeBot implements StockObserver {
    private final double threshold;
    public TradeBot(double threshold) { this.threshold = threshold; }
    public void onPriceUpdate(String symbol, double price) {
        String action = price > threshold ? "SELL signal" : "BUY signal";
        System.out.println("[TradeBot] " + symbol +  @ " + price +  vs threshold " + threshold +  -> " + action);
    }
}

// ---------- StockData.java (Subject) ----------
import java.util.*;
public class StockData {
    private List<StockObserver> observers = new ArrayList<>();

    public void addObserver(StockObserver o)    { observers.add(o); }
    public void removeObserver(StockObserver o) { observers.remove(o); }

    public void setPrice(String symbol, double price) {
        System.out.println("\n[StockData] " + symbol +  updated to $" + price);
        for (StockObserver o : observers) o.onPriceUpdate(symbol, price);
    }
}

// ---------- Demo.java ----------
public class Demo {
    public static void main(String[] args) {
        StockData feed = new StockData();
        StockObserver ticker = new TickerTape();
        StockObserver graph  = new GraphWidget();
        StockObserver bot    = new TradeBot(700.0);

        feed.addObserver(ticker);
        feed.addObserver(graph);
        feed.addObserver(bot);
        feed.setPrice("AAPL", 150.25);

        feed.removeObserver(bot);           // bot detaches -- runtime removal
        feed.setPrice("AAPL", 151.10);

        feed.addObserver(bot);              // bot re-attaches
        feed.setPrice("TSLA", 705.50);
    }
}
